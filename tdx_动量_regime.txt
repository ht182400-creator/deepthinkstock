{ ============================================================
  周频 动量+Regime 策略 · 通达信【周线】技术指标公式
  ------------------------------------------------------------
  用法：
    1) 通达信 → Ctrl+F → 公式管理器 → 技术指标公式 → 新建
    2) 名称自取（如 ZL_MOMENTUM_REGIME），粘贴本文件全部内容
    3) 在个股 K 线图切到【周线】周期，输入公式名调用
  ------------------------------------------------------------
  与「质量+Regime」的区别：纯动量版【不】做基本面质量门，
  只按 趋势/动量/Regime 给出买卖点。横截面排名（选前 N 名）
  仍需配合外部候选池清单使用，单股指标无法复现。
  ------------------------------------------------------------
  图标：买点=红箭头(8)  卖点=绿箭头(9)  量能蓄势=小三角(3)
  ============================================================ }

N_MOM:=52;      { 动量 / 趋势窗口 = 52 周 }
N_REG:=56;      { Regime：上证指数 56 周均线 }
N_VOL:=26;      { 量能窗口 = 26 周 }

{ —— Regime：以上证指数 56 周均线判断大盘多空 —— }
{ 若你的通达信不支持跨品种引用 "999999$CLOSE"，可改为 INDEXC
  （自动取个股对应市场指数；严格对齐回测请用上证 999999） }
SHC:="999999$CLOSE";
SH_MA56:=MA(SHC,N_REG);
REGIME_UP:=SHC>SH_MA56;        { 大盘多头 }
EMPTY:=NOT REGIME_UP;          { 大盘空头 → 强制空仓 }

{ —— 个股趋势 / 动量（52 周） —— }
MA52:=MA(C,N_MOM);
MOM52:=C/REF(C,N_MOM)-1;       { 52 周收益率 }
PRICE_ABOVE:=C>MA52;           { 站上 52 周均线 }
MOM_POS:=MOM52>0;              { 动量正向 }

{ —— 量能（可视化参考，不进买点条件） —— }
OBV_LINE:=OBV;
OBV_MA:=MA(OBV_LINE,N_VOL);
VOL_MA:=MA(VOL,N_VOL);
VOL_OK:=(OBV_LINE>OBV_MA) AND (VOL>VOL_MA);   { 资金蓄势 + 放量 }

{ —— 买点：Regime 多头 + 站上 52 周线 + 动量正向（纯动量，无质量门） —— }
BUY:=REGIME_UP AND PRICE_ABOVE AND MOM_POS;
FRESH_BUY:=BUY AND NOT REF(BUY,1);

{ —— 卖点 / 退出：破 52 周线 或 动量≤-40%
       或 估值分位≈顶部(HHV260 的 85%) 或 Regime 翻空 —— }
HIGH_VAL:=C>=HHV(C,260)*0.85;
SELL:=(C<MA52) OR (MOM52<=-0.40) OR HIGH_VAL OR EMPTY;
FRESH_SELL:=SELL AND NOT REF(SELL,1);

{ —— 绘图（叠加在周 K 线上） —— }
MA52:MA52,COLORORANGE,LINETHICK2;
DRAWGBK(EMPTY,RGB(255,222,222));                          { 空仓：背景淡红 }
DRAWTEXT(EMPTY,HIGH*1.04,'空仓'),COLORRED;                { 空仓文字标注 }
DRAWICON(FRESH_BUY,LOW*0.98,8);                           { 买点：红箭头 }
DRAWICON(FRESH_SELL,HIGH*1.02,9);                         { 卖点：绿箭头 }
DRAWICON(VOL_OK,LOW*0.97,3);                              { 量能蓄势：小三角（仅参考） }
